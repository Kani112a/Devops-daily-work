}




Kubernetes code for jenkins:
pipeline {
    agent any

    environment {
        IMAGE_NAME = 'devops-nginx:jenkins'
        POD_NAME   = 'nginx-pod'
        KUBE_NAMESPACE = 'default'  // change if you use another namespace
    }

    stages {

        stage('Checkout Code') {
            steps {
                git branch: 'main', url: 'https://github.com/Kani112a/Devops-daily-work.git'
            }
        }

        stage('Build Docker Image') {
            steps {
                bat "docker build -t ${IMAGE_NAME} ./docker"
            }
        }

        stage('Push Docker Image') {
            steps {
                // Optional: if using Docker Hub or private registry
                // bat "docker tag ${IMAGE_NAME} yourdockerhubusername/${IMAGE_NAME}"
                // bat "docker push yourdockerhubusername/${IMAGE_NAME}"
                echo "Skipping push if using local Docker"
            }
        }

        stage('Deploy to Kubernetes') {
            steps {
                // Delete old pod if exists
                bat "kubectl delete pod ${POD_NAME} --ignore-not-found=true --namespace=${KUBE_NAMESPACE}"

                // Apply pod.yaml or create pod using inline manifest
                writeFile file: 'k8s-pod.yaml', text: """
apiVersion: v1
kind: Pod
metadata:
  name: ${POD_NAME}
  namespace: ${KUBE_NAMESPACE}
spec:
  containers:
  - name: nginx
    image: ${IMAGE_NAME}
    ports:
    - containerPort: 80
"""
                // Apply the pod manifest
                bat "kubectl apply -f k8s-pod.yaml"
            }
        }

        stage('Verify Pod') {
            steps {
                bat "kubectl get pods -n ${KUBE_NAMESPACE}"
            }
        }
    }
}    ----Not used just a reference


KUBERNETES DEPLOYMENT & REPLICASET:
1.Create Deployments
2.Scale with ReplicaSets
3.Understand self-healing & rolling updates
Create file: vim deployment.yaml
Delpoyment.yaml content:
apiVersion: apps/v1
kind: Deployment
metadata:
  name: nginx-deployment
spec:
  replicas: 3
  selector:
    matchLabels:
      app: nginx
  template:
    metadata:
      labels:
        app: nginx
    spec:
      containers:
      - name: nginx
        image: devops-nginx:jenkins
        imagePullPolicy: Never
        ports:
        - containerPort: 80

kubectl apply -f kubernetes/deployment.yaml
kubectl get deployments
kubectl get pods

Scale Deployment:
kubectl scale deployment nginx-deployment --replicas=5
kubectl get pods
----Pods will auto-scale.

Rolling Update:
Update image (simulate version upgrade):
kubectl set image deployment/nginx-deployment nginx=nginx:alpine
kubectl rollout status deployment/nginx-deployment
Deployment manages replicas of Pods.
ReplicaSet ensures desired number of Pods.
Rolling updates allow zero-downtime updates.
